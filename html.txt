<script>
const LADDERS = { 4:25, 13:46, 33:49, 42:63, 50:69, 62:81, 74:92 };
const SNAKES  = { 27:5, 40:3, 54:31, 66:45, 76:58, 89:53, 99:41 };
const totalCells = 100;

const board = document.getElementById('board');
const msgEl = document.getElementById('msg');
const dieEl = document.getElementById('die');
const p1El  = document.getElementById('p1');
const p2El  = document.getElementById('p2');

let pos = { 1: 1, 2: 1 };
let turn = 1;          // 1 = أحمد خالد ، 2 = الحاسوب
let rolling = false;

// بناء اللوحة من 100 لـ 1 (الصفوف من تحت لفوق)
function buildBoard() {
  board.innerHTML = '';
  for (let row = 9; row >= 0; row--) {
    const start = row * 10;
    let cols = [];
    for (let i = 1; i <= 10; i++) cols.push(start + i);
    if (row % 2 === 1) cols.reverse(); // snake pattern
    cols.forEach(n => {
      const c = document.createElement('div');
      c.className = 'cell';
      c.id = 'cell-' + n;
      if (LADDERS[n]) c.classList.add('ladder');
      if (SNAKES[n])  c.classList.add('snake');
      c.innerHTML = `<span class="num">${n}</span><span class="tokens" id="tok-${n}"></span>`;
      board.appendChild(c);
    });
  }
  render();
}

function render() {
  document.querySelectorAll('.tokens').forEach(t => t.innerHTML = '');
  const t1 = document.createElement('div'); t1.textContent = '🧑';
  const t2 = document.createElement('div'); t2.textContent = '🤖';
  const c1 = document.getElementById('tok-' + pos[1]);
  const c2 = document.getElementById('tok-' + pos[2]);
  if (c1) c1.appendChild(t1);
  if (c2) c2.appendChild(t2);
  p1El.textContent = pos[1];
  p2El.textContent = pos[2];
}

function move(player, steps) {
  let target = pos[player] + steps;
  if (target > 100) {
    msgFor(player, `رميت ${steps} بس محتاج رقم مظبوط توصل لـ 100 بالظبط 😅`);
    return false;
  }
  pos[player] = target;

  if (LADDERS[target]) {
    msgFor(player, `رميت ${steps} وطلعت على سلم! 🪜 من ${target} لـ ${LADDERS[target]} — طلوع يا معلم!`);
    pos[player] = LADDERS[target];
  } else if (SNAKES[target]) {
    msgFor(player, `رميت ${steps} ووقعت على تعبان! 🐍 من ${target} نزلت لـ ${SNAKES[target]} — يا خبر!`);
    pos[player] = SNAKES[target];
  } else {
    msgFor(player, `رميت ${steps} ووصلت خانة ${target}`);
  }

  render();

  if (pos[player] === 100) {
    msgEl.textContent = player === 1
      ? '🎉 مبروك يا أحمد خالد! كسبت اللعبة يا بطل 🇪🇬'
      : '😢 الحاسوب كسب المرة دي.. جرب تاني يا أحمد!';
    document.getElementById('rollBtn').disabled = true;
    return true;
  }
  return false;
}

function msgFor(player, text) {
  msgEl.textContent = (player === 1 ? '🧑 أحمد خالد: ' : '🤖 الحاسوب: ') + text;
}

function rollDie() {
  return Math.floor(Math.random() * 6) + 1;
}

document.getElementById('rollBtn').addEventListener('click', () => {
  if (rolling) return;
  rolling = true;
  const btn = document.getElementById('rollBtn');
  btn.disabled = true;

  const val = rollDie();
  dieEl.textContent = '🎲';
  let count = 0;
  const spin = setInterval(() => {
    dieEl.textContent = '🎲 ' + (Math.floor(Math.random()*6)+1);
    if (++count > 8) {
      clearInterval(spin);
      dieEl.textContent = '🎲 ' + val;
      const won = move(1, val);
      if (won) { rolling = false; return; }
      turn = 2;
      setTimeout(computerTurn, 900);
    }
  }, 70);
});

function computerTurn() {
  const val = rollDie();
  dieEl.textContent = '🎲 ' + val;
  const won = move(2, val);
  if (won) { rolling = false; return; }
  turn = 1;
  rolling = false;
  document.getElementById('rollBtn').disabled = false;
}

document.getElementById('resetBtn').addEventListener('click', () => {
  pos = { 1: 1, 2: 1 };
  turn = 1; rolling = false;
  dieEl.textContent = '🎲';
  document.getElementById('rollBtn').disabled = false;
  msgEl.textContent = 'يلا نلعب من جديد يا أحمد خالد! 🇪🇬';
  render();
});

buildBoard();
